import socket
with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as client:
    client.settimeout(1)
    client.connect((foreign_host, 113))
    client.send(f"{foreign_port},{local_port}\r\n".encode())
    result = client.recv(4096).decode().split(":")[-1].strip()
