line.py
    import re, sys
    for line in sys.stdin:
        m = re.match(r'((?P<heads>(?P<h3>###)|(?P<h2>##)|(?P<h1>#))|(?P<list>\* )|(?P<link>=> (?P<url>[^\s]+))|(?P<quote>>))\s*(?P<content>.*)$', line)
        if m:
            print(m.groupdict())
