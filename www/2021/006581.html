With nginx it's quite simple, just ensure that your nginx is compiled
with stream_ssl_preread_module:

stream {
    map $ssl_preread_server_name $upstream {
        "app1.example.com" 127.0.0.1:5000;
        "app2.example.com" 127.0.0.1:5001;
    }
    server {
        listen      1965;
        proxy_pass  $upstream;
        ssl_preread on;
    }
}
